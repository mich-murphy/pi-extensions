"""Development-only retrospective diagnostic. No API, receipt, policy or ledger writes."""
from pathlib import Path
import json, math, os
r=Path.home()/'.pi/agent/pi-terminology/recall-2026-10-22'
rows=json.loads((r/'fixtures/calibration-consensus-selected-v2.json').read_text())
obs={x['observation']['id']:x['observation'] for x in map(json.loads,(r/'calibrate.observations.jsonl').read_text().splitlines())}
old=json.loads((r/'calibrate.report.json').read_text())['thresholds']
ids=sorted(old)
counts={i:sum(x['labels'][i]=='positive' and i not in x['named'] for x in rows) for i in ids}
def per(i,t):
 tp=fp=ah=0
 for x in rows:
  o=obs[x['id']]
  if i in o['alreadyNamed'] or o['probabilities'][i]['native']<t: continue
  if x['labels'][i]=='positive' and i not in x['named']: tp+=1
  elif x['labels'][i]=='negative' or i in x['named']: fp+=1
  else: ah+=1
 return tp,fp,ah
options={}
curves={}
for i in ids:
 groups={}
 for n in range(80,100):
  t=n/100;tp,fp,ah=per(i,t)
  if i in ['core:refactoring','core:low-coupling','core:separation-of-concerns']:
   curves.setdefault(i,[]).append({'t':t,'tp':tp,'fp':fp,'recall':tp/counts[i],'precision':tp/(tp+fp) if tp+fp else None})
  if tp/counts[i]>=.7:
   groups.setdefault((tp,fp),[]).append(n)
 options[i]=[{'t':sorted(ns,key=lambda n:(abs(n-(min(ns)+max(ns))/2),-n))[0]/100,'tp':k[0],'fp':k[1]} for k,ns in groups.items()]
missing=[i for i in ids if not options[i]]
# Exact multiple-choice raw-score knapsack frontier. Widget coupling checked independently below.
bound=sum(counts.values())//9
states={0:(0,{})}
for i in ids:
 new={}
 for f,(p,ts) in states.items():
  for c in options[i]:
   nf=f+c['fp'];np=p+c['tp']
   if nf<=bound and (nf not in new or np>new[nf][0]): new[nf]=(np,{**ts,i:c['t']})
 states=new
feasible=[(p,f,ts) for f,(p,ts) in states.items() if p>=9*f]
result={'developmentOnly':True,'promotable':False,'candidateFloor':.8,'missingRecallOptions':missing,'totalPositiveAssignments':sum(counts.values()),'curves':curves}
if feasible:
 tp,fp,thresholds=max(feasible,key=lambda v:(v[0],-v[1]))
 cls={i:{'tp':0,'fp':0,'fn':counts[i],'ambiguousHits':0} for i in ids}
 hidden=0
 for x in rows:
  o=obs[x['id']]
  selected=[i for i in ids if i not in o['alreadyNamed'] and o['probabilities'][i]['native']>=thresholds[i]]
  hidden+=sum(x['labels'][i]=='positive' and i not in x['named'] for i in selected[3:])
  for i in selected[:3]:
   if x['labels'][i]=='positive' and i not in x['named']:cls[i]['tp']+=1;cls[i]['fn']-=1
   elif x['labels'][i]=='negative' or i in x['named']:cls[i]['fp']+=1
   else:cls[i]['ambiguousHits']+=1
 for i,c in cls.items():
  c['recall']=c['tp']/counts[i];c['precision']=c['tp']/(c['tp']+c['fp']) if c['tp']+c['fp'] else None
 vt=sum(c['tp'] for c in cls.values());vf=sum(c['fp'] for c in cls.values());va=sum(c['ambiguousHits'] for c in cls.values())
 result.update({'raw':{'tp':tp,'fp':fp,'precision':tp/(tp+fp),'recall':tp/sum(counts.values())},'displayed':{'tp':vt,'fp':vf,'fn':sum(counts.values())-vt,'ambiguousHits':va,'precision':vt/(vt+vf),'recall':vt/sum(counts.values()),'conservativePrecision':vt/(vt+vf+va),'hiddenPositiveAssignments':hidden},'thresholds':thresholds,'perConcept':cls,'visibleRecallFailures':[i for i,c in cls.items() if c['recall']<.7]})
p=r/'offline-aggregate-frontier.json'
fd=os.open(p,os.O_WRONLY|os.O_CREAT|os.O_EXCL,0o600)
with os.fdopen(fd,'w') as f:json.dump(result,f,indent=2);f.write('\n');f.flush();os.fsync(f.fileno())
print(json.dumps({k:v for k,v in result.items() if k not in ['curves','perConcept','thresholds']},indent=2))
print('Changed thresholds:',json.dumps({i:{'from':old[i],'to':result.get('thresholds',{}).get(i)} for i in ids if old[i]!=result.get('thresholds',{}).get(i)},indent=2))
print('Failed-class curves:',json.dumps(curves,indent=2))
